## Technical Details

### Connection Test Script
Create a script `scripts/test_db_connection.py`:
```python
import os
import sys
from sqlalchemy import create_engine, text

def test_connection(db_url: str, db_name: str):
    """Test database connection and basic operations"""
    try:
        engine = create_engine(db_url)
        with engine.connect() as conn:
            result = conn.execute(text("SELECT version();"))
            version = result.fetchone()[0]
            print(f"✅ {db_name} connection successful!")
            print(f"   PostgreSQL version: {version}")
            return True
    except Exception as e:
        print(f"❌ {db_name} connection failed: {str(e)}")
        return False

if __name__ == "__main__":
    dev_url = os.getenv("DATABASE_URL")
    prod_url = os.getenv("DATABASE_URL_PROD")
    
    print("Testing database connections...\n")
    dev_ok = test_connection(dev_url, "Development DB")
    prod_ok = test_connection(prod_url, "Production DB")
    
    if not dev_ok or not prod_ok:
        sys.exit(1)
```
